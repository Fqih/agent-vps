> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Hub: ntfy push, a `/status` endpoint for a KWGT phone widget, a 24/7 LLM task queue over 9router, and a Google Calendar connector, wired into the existing infobot.

**Architecture:** New packages `hub/` and `connectors/` next to the existing flat modules in this repo (`tg.py`, `digest.py`, `monitor.py`, ...). Two new systemd services (`hub-worker`, `hub-status`) and one Docker container (ntfy). Everything binds to `127.0.0.1` and is published to the tailnet with `tailscale serve`.

**Tech Stack:** Python 3.8 stdlib only (urllib, sqlite3, http.server, unittest), SQLite (WAL), systemd, Docker (ntfy), KWGT on Android.

**Spec:** `docs/superpowers/specs/2026-10-16-hub-design.md`

## Global Constraints

- Runtime is Python 3.8.10 on the VPS. No `tomllib`, no `match`, no `str.removeprefix`, no `zoneinfo`. Every module starts with `from __future__ import annotations`.
- Stdlib only. No new Python dependencies.
- Tests use `unittest` (repo convention: `python3 -m unittest ...`), file names `test_*.py` in the repo root.
- Run every command from the repo root (`~/Server Agent/agent-vps`); modules are imported as `hub.x`, `connectors.x`, and the existing flat modules (`tg`, `digest`, `monitor`, `tracker`) by bare name.
- Services bind to `127.0.0.1` only: status on port `8090`, ntfy on `8091`.
- Model chains live in `hub/models.json`. `cc/` (Claude subscription) is never used. Antigravity Claude 4.6 / gpt-oss only appear in the `bulk` chain.
- No secrets, emails or detailed application content in `/status`.
- ntfy topics: `alerts` (high), `jobs`, `tasks` (default), `digest` (low).
- Commits: conventional style, author `Fqih <mhmdfkih21@gmail.com>`, no `Co-Authored-By` trailer, no generated-with footer. Do not push.
- This repo is public. Never commit tokens, IPs of the tailnet, or emails.

## Review Focus

- Worker restarted while a task is `running`: the task must run again right after restart, not wait 15 minutes. Pinned in Task 2 (`test_requeue_running_on_start`).
- 9router reports a missing model as HTTP 503 whose body contains `] [404]:`; that model must still be skipped for 6 hours. Pinned in Task 3 (`test_wrapped_404_marks_dead`).
- Indonesian text and emoji in titles and prompts (`"Lamar PT Ölçü 🎯"`) must reach ntfy and the queue intact. Pinned in Task 1 and Task 2.
- `STATUS_TOKEN` missing or empty must deny every request, never allow all. Pinned in Task 7 (`test_empty_token_denies`).
- Before the owner logs in to Google, the widget shows `calendar: "off"` and `level` stays `ok`, not `warn`. Pinned in Task 6 (`test_no_token_is_off_not_error`).

```python
# hub/notify.py
"""Push notifications through the self-hosted ntfy server (Telegram fallback for high priority)."""
from __future__ import annotations

import json
import logging
import urllib.error
import urllib.request
from html import escape

import tg

log = logging.getLogger("hub.notify")
PRIORITY = {"high": 5, "default": 3, "low": 2}


def publish(env: dict, topic: str, title: str, message: str, priority: str = "default",
            click: str | None = None, tags: list | None = None, fallback: bool = True) -> bool:
    """Send one push. Never raises; True when ntfy accepted it."""
    body = {"topic": topic, "title": title, "message": message, "priority": PRIORITY.get(priority, 3)}
    if click:
        body["click"] = click
    if tags:
        body["tags"] = tags
    url = env.get("NTFY_URL", "").rstrip("/")
    try:
        if not url:
            raise ValueError("NTFY_URL not set")
        req = urllib.request.Request(url, data=json.dumps(body).encode(), headers={
            "Content-Type": "application/json", "Authorization": f"Bearer {env.get('NTFY_TOKEN', '')}"})
        with urllib.request.urlopen(req, timeout=15):
            return True
    except (urllib.error.URLError, OSError, ValueError) as err:
        log.warning("ntfy %s gagal: %s", topic, err)
    if priority == "high" and fallback:
        try:
            tg.send(env, f"<b>{escape(title)}</b>\n{escape(message)}")
        except Exception:  # the fallback must not take the caller down either
            log.exception("telegram fallback gagal")
    return False
```

```python
# hub/tasks.py
"""Task queue in its own SQLite file so long LLM work never locks state.sqlite."""
from __future__ import annotations

import json
import sqlite3
from datetime import datetime, timedelta, timezone
from pathlib import Path

DB_PATH = Path(__file__).resolve().parent.parent / "tasks.sqlite"
RETRY_DELAYS = (timedelta(minutes=1), timedelta(minutes=5), timedelta(minutes=30))
SCHEMA = """
CREATE TABLE IF NOT EXISTS tasks (
    id INTEGER PRIMARY KEY,
    kind TEXT NOT NULL,
    payload TEXT NOT NULL,
    chain TEXT NOT NULL DEFAULT 'fast',
    dedup_key TEXT UNIQUE,
    status TEXT NOT NULL,
    attempts INTEGER NOT NULL DEFAULT 0,
    run_after TEXT NOT NULL,
    started_at TEXT,
    finished_at TEXT,
    model TEXT,
    result TEXT,
    error TEXT,
    created_at TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS tasks_ready ON tasks (status, run_after);
"""


def utcnow() -> datetime:
    return datetime.now(timezone.utc)


def iso(dt: datetime) -> str:
    return dt.astimezone(timezone.utc).isoformat(timespec="seconds")


def connect(path=DB_PATH) -> sqlite3.Connection:
    db = sqlite3.connect(str(path), timeout=30, check_same_thread=False)
    db.row_factory = sqlite3.Row
    db.execute("PRAGMA journal_mode=WAL")
    db.executescript(SCHEMA)
    return db


def enqueue(db: sqlite3.Connection, kind: str, payload: dict, chain: str = "fast",
            dedup_key: str | None = None, now: datetime | None = None) -> int:
    if dedup_key:
        row = db.execute("SELECT id FROM tasks WHERE dedup_key = ?", (dedup_key,)).fetchone()
        if row:
            return row["id"]
    stamp = iso(now or utcnow())
    cur = db.execute(
        "INSERT INTO tasks (kind, payload, chain, dedup_key, status, run_after, created_at) "
        "VALUES (?, ?, ?, ?, 'queued', ?, ?)",
        (kind, json.dumps(payload, ensure_ascii=False), chain, dedup_key, stamp, stamp))
    db.commit()
    return cur.lastrowid


def get(db: sqlite3.Connection, task_id: int) -> dict | None:
    row = db.execute("SELECT * FROM tasks WHERE id = ?", (task_id,)).fetchone()
    if row is None:
        return None
    task = dict(row)
    task["payload"] = json.loads(task["payload"])
    return task


def claim(db: sqlite3.Connection, now: datetime | None = None) -> dict | None:
    stamp = iso(now or utcnow())
    row = db.execute("SELECT id FROM tasks WHERE status = 'queued' AND run_after <= ? "
                     "ORDER BY run_after, id LIMIT 1", (stamp,)).fetchone()
    if row is None:
        return None
    db.execute("UPDATE tasks SET status = 'running', started_at = ? WHERE id = ?", (stamp, row["id"]))
    db.commit()
    return get(db, row["id"])


def finish(db: sqlite3.Connection, task_id: int, result: str, model: str, now: datetime | None = None) -> None:
    db.execute("UPDATE tasks SET status = 'done', result = ?, model = ?, error = NULL, finished_at = ? "
               "WHERE id = ?", (result, model, iso(now or utcnow()), task_id))
    db.commit()


def fail_now(db: sqlite3.Connection, task_id: int, error: str, now: datetime | None = None) -> None:
    db.execute("UPDATE tasks SET status = 'failed', error = ?, finished_at = ? WHERE id = ?",
               (error[:500], iso(now or utcnow()), task_id))
    db.commit()


def retry_or_fail(db: sqlite3.Connection, task_id: int, error: str, now: datetime | None = None) -> str:
    now = now or utcnow()
    attempts = db.execute("SELECT attempts FROM tasks WHERE id = ?", (task_id,)).fetchone()["attempts"] + 1
    if attempts > len(RETRY_DELAYS):
        db.execute("UPDATE tasks SET status = 'failed', attempts = ?, error = ?, finished_at = ? WHERE id = ?",
                   (attempts, error[:500], iso(now), task_id))
        db.commit()
        return "failed"
    db.execute("UPDATE tasks SET status = 'queued', attempts = ?, error = ?, run_after = ?, started_at = NULL "
               "WHERE id = ?", (attempts, error[:500], iso(now + RETRY_DELAYS[attempts - 1]), task_id))
    db.commit()
    return "queued"


def requeue_running(db: sqlite3.Connection) -> int:
    """Single worker: anything still 'running' at start-up was orphaned by a crash."""
    cur = db.execute("UPDATE tasks SET status = 'queued', started_at = NULL WHERE status = 'running'")
    db.commit()
    return cur.rowcount


def counts(db: sqlite3.Connection, now: datetime | None = None) -> dict:
    now = now or utcnow()
    live = {status: n for status, n in db.execute(
        "SELECT status, COUNT(*) FROM tasks WHERE status IN ('queued', 'running') GROUP BY status")}
    failed = db.execute("SELECT COUNT(*) FROM tasks WHERE status = 'failed' AND finished_at >= ?",
                        (iso(now - timedelta(days=1)),)).fetchone()[0]
    return {"running": live.get("running", 0), "queued": live.get("queued", 0), "failed": failed}


def recent(db: sqlite3.Connection, limit: int = 20) -> list:
    rows = db.execute(
        "SELECT id, kind, status, model, created_at, finished_at, payload, "
        "substr(coalesce(result, ''), 1, 200) AS preview, error FROM tasks ORDER BY id DESC LIMIT ?", (limit,))
    out = []
    for row in rows:
        item = dict(row)
        item["title"] = json.loads(item.pop("payload")).get("title") or item["kind"]
        out.append(item)
    return out
```

```python
# hub/llm.py
"""Chat over 9router with an ordered model chain and a short memory of dead models."""
from __future__ import annotations

import json
import logging
import socket
import urllib.error
import urllib.request
from datetime import datetime, timedelta, timezone
from pathlib import Path

MODELS_PATH = Path(__file__).with_name("models.json")
DEAD_FOR = timedelta(hours=6)
TIMEOUT = 180
log = logging.getLogger("hub.llm")
_dead: dict = {}


class ChainFailed(RuntimeError):
    """Every model in the chain failed or is marked dead."""


def load_chains(path=MODELS_PATH) -> dict:
    return json.loads(Path(path).read_text())


def _not_found(code: int, body: str) -> bool:
    # 9router answers 503 and puts the upstream status in the message: "[provider/model] [404]: ..."
    return code == 404 or "] [404]:" in body


def _call(model: str, messages: list, cfg: dict) -> str:
    url = cfg.get("base_url", "http://127.0.0.1:20128/v1").rstrip("/") + "/chat/completions"
    req = urllib.request.Request(
        url, data=json.dumps({"model": model, "stream": False, "messages": messages}).encode(),
        headers={"Content-Type": "application/json", "Authorization": f"Bearer {cfg.get('api_key', '')}"})
    with urllib.request.urlopen(req, timeout=TIMEOUT) as resp:
        data = json.loads(resp.read().decode("utf-8", "replace"))
    text = (data["choices"][0]["message"]["content"] or "").strip()
    if not text:
        raise ValueError("empty answer")
    return text


def chat(chain: str, messages: list, cfg: dict, chains: dict | None = None,
         now: datetime | None = None) -> tuple:
    now = now or datetime.now(timezone.utc)
    errors = []
    for model in (chains or load_chains())[chain]:
        if _dead.get(model, now) > now:
            continue
        try:
            return _call(model, messages, cfg), model
        except urllib.error.HTTPError as err:
            body = err.read().decode("utf-8", "replace")
            if _not_found(err.code, body):
                _dead[model] = now + DEAD_FOR
            errors.append(f"{model}: HTTP {err.code}")
        except (urllib.error.URLError, socket.timeout, OSError, KeyError, IndexError, ValueError) as err:
            errors.append(f"{model}: {err}")
        log.warning("model gagal: %s", errors[-1])
    raise ChainFailed("; ".join(errors) or f"semua model di chain {chain} sedang ditandai mati")
```

```python
# hub/worker.py
#!/usr/bin/env python3
"""24/7 task worker: one task at a time through 9router, ntfy when done. Run: python3 -m hub.worker"""
from __future__ import annotations

import logging
import time
from datetime import datetime

from digest import REPORT_DIR, load_env, model_config
from hub import notify, tasks
from hub.handlers import HANDLERS

POLL_SECONDS = 5
log = logging.getLogger("hub.worker")


def task_link(ctx: dict, task_id: int) -> str | None:
    base = ctx["env"].get("STATUS_PUBLIC_URL", "").rstrip("/")
    return f"{base}/task/{task_id}?token={ctx['env'].get('STATUS_TOKEN', '')}" if base else None


def run_once(db, ctx: dict, now: datetime | None = None) -> bool:
    task = tasks.claim(db, now)
    if task is None:
        return False
    title = task["payload"].get("title") or task["kind"]
    handler = HANDLERS.get(task["kind"])
    if handler is None:
        tasks.fail_now(db, task["id"], f"jenis tugas tidak dikenal: {task['kind']}", now)
        return True
    try:
        text, model = handler(task, ctx)
    except Exception as err:  # any handler error goes through the retry schedule
        log.warning("tugas #%s gagal: %s", task["id"], err)
        if tasks.retry_or_fail(db, task["id"], str(err), now) == "failed":
            notify.publish(ctx["env"], "tasks", f"Gagal: {title}", str(err)[:300],
                           click=task_link(ctx, task["id"]))
        return True
    tasks.finish(db, task["id"], text, model, now)
    if task["payload"].get("notify", True):
        notify.publish(ctx["env"], "tasks", f"Selesai: {title}", text[:400], click=task_link(ctx, task["id"]))
    return True


def main() -> None:
    logging.basicConfig(filename=REPORT_DIR / "hub-worker.log", level=logging.INFO,
                        format="[%(asctime)s] %(levelname)s %(name)s %(message)s")
    env = load_env()
    ctx = {"env": env, "cfg": model_config(env)}
    db = tasks.connect()
    log.info("worker jalan, %d tugas dikembalikan ke antrean", tasks.requeue_running(db))
    while True:
        try:
            if not run_once(db, ctx):
                time.sleep(POLL_SECONDS)
        except Exception:
            log.exception("worker loop error")
            time.sleep(POLL_SECONDS)


if __name__ == "__main__":
    main()
```

```python
class AuthError(RuntimeError):
    """Google rejected the token; the owner must log in again."""


class TokenMissing(AuthError):
    """No token file yet (Google login never done)."""


def _post(url: str, fields: dict) -> dict:
    req = urllib.request.Request(url, data=urllib.parse.urlencode(fields).encode())
    with urllib.request.urlopen(req, timeout=30) as resp:
        return json.loads(resp.read())


def _save(token: dict, path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(".tmp")
    tmp.write_text(json.dumps(token))
    os.chmod(tmp, 0o600)
    tmp.replace(path)


def access_token(path=TOKEN_PATH, now: datetime | None = None) -> str:
    now = now or datetime.now(timezone.utc)
    path = Path(path)
    try:
        token = json.loads(path.read_text())
    except FileNotFoundError as err:
        raise TokenMissing("belum login Google") from err
    except (OSError, ValueError) as err:
        raise AuthError(f"token Google rusak: {err}") from err
    expiry = datetime.fromisoformat(token["expiry"]) if token.get("expiry") else now
    if token.get("access_token") and expiry - now > REFRESH_MARGIN:
        return token["access_token"]
    try:
        fresh = _post(TOKEN_URL, {"client_id": token["client_id"], "client_secret": token["client_secret"],
                                  "refresh_token": token["refresh_token"], "grant_type": "refresh_token"})
    except urllib.error.HTTPError as err:
        if "invalid_grant" in err.read().decode("utf-8", "replace"):
            raise AuthError("refresh token Google dicabut atau kedaluwarsa") from err
        raise
    updated = {**token, "access_token": fresh["access_token"],
               "expiry": (now + timedelta(seconds=int(fresh.get("expires_in", 3600)))).isoformat()}
    _save(updated, path)
    return updated["access_token"]

            self.send_response(200)
            self.end_headers()
            self.wfile.write(b"Login selesai, tab ini boleh ditutup.")

            pass
```

```python
"""Google Calendar (primary calendar) over REST."""
from __future__ import annotations

import json
import urllib.parse
import urllib.request
from datetime import date, datetime, time as dtime, timedelta, timezone

from connectors.google_auth import access_token

API = "https://www.googleapis.com/calendar/v3/calendars/primary/events"
WIB = timezone(timedelta(hours=7))


def _request(method: str, url: str, token: str, body: dict | None = None) -> dict:
    req = urllib.request.Request(url, method=method, data=json.dumps(body).encode() if body is not None else None,
                                 headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=30) as resp:
        return json.loads(resp.read())


def _event(item: dict) -> dict:
    raw = item.get("start", {}).get("dateTime")
    start = datetime.fromisoformat(raw.replace("Z", "+00:00")).astimezone(WIB) if raw else None
    return {"id": item.get("id"), "title": item.get("summary") or "(tanpa judul)",
            "start": start.isoformat() if start else None, "time": start.strftime("%H:%M") if start else None}


def list_events(day: date, token_fn=access_token) -> list:
    start = datetime.combine(day, dtime.min, WIB)
    query = urllib.parse.urlencode({"timeMin": start.isoformat(), "timeMax": (start + timedelta(days=1)).isoformat(),
                                    "singleEvents": "true", "orderBy": "startTime", "maxResults": "20"})
    data = _request("GET", f"{API}?{query}", token_fn())
    return [_event(item) for item in data.get("items", [])]


def create_event(summary: str, start: datetime, end: datetime, description: str = "",
                 token_fn=access_token) -> str:
    body = {"summary": summary, "description": description,
            "start": {"dateTime": start.isoformat()}, "end": {"dateTime": end.isoformat()}}
```

```python
"""VPS health: load, RAM, disk, failing monitor checks, last backup."""
from __future__ import annotations

import os
import re
import shutil
from contextlib import closing
from datetime import datetime, timedelta
from pathlib import Path

def _ram_percent(meminfo: Path) -> int:
    mem = dict(re.findall(r"(\w+):\s+(\d+) kB", meminfo.read_text()))
    return round(100 - int(mem["MemAvailable"]) * 100 / int(mem["MemTotal"]))


def _backup_ok(dest: Path, now: datetime) -> bool:
    """A snapshot from today or yesterday (WIB) exists; the 03:00 run makes today's."""
    day = now.astimezone(WIB).date()
    return any(any((dest / d.isoformat()).glob("*.sqlite.gz")) for d in (day, day - timedelta(days=1)))


def collect(ctx: dict, now: datetime) -> dict:
    load1 = float(Path(ctx.get("loadavg", "/proc/loadavg")).read_text().split()[0])
    disk = shutil.disk_usage("/")
    with closing(ro(ctx["state_db"])) as db:
        down = [name for (name,) in db.execute("SELECT name FROM checks WHERE ok = 0 ORDER BY name")]
    return {
        "cpu": min(100, round(load1 / (os.cpu_count() or 1) * 100)),
        "ram": _ram_percent(Path(ctx.get("meminfo", "/proc/meminfo"))),
        "disk": disk.used * 100 // disk.total,
        "down": down,
        "backup_ok": _backup_ok(Path(ctx["backups"]), now),
    }
```

```python
CACHE_SECONDS = 60

              "vps": vps.collect, "router": router9.collect}
LEVEL_ICON = {"ok": "🟢", "warn": "🟡", "crit": "🔴"}
log = logging.getLogger("hub.status")
_cache: dict = {}


def level(s: dict) -> str:
    v = s.get("vps", {})
    if any(name.startswith("container:") for name in v.get("down", [])) or v.get("backup_ok") is False:
        return "crit"
    warn = (v.get("disk", 0) > 85 or v.get("ram", 0) > 90 or bool(v.get("down"))
            or s.get("queue", {}).get("failed", 0) > 0
            or str(s.get("today", {}).get("calendar", "ok")).startswith("error")
            or any(isinstance(part, dict) and "error" in part for part in s.values()))
    return "warn" if warn else "ok"


def gather(ctx: dict, now: datetime, collectors: dict | None = None, clock=time.monotonic) -> dict:
    out = {"updated": now.astimezone(WIB).isoformat(timespec="seconds")}
    for name, fn in (collectors if collectors is not None else COLLECTORS).items():
        hit = _cache.get(name)
        if hit and clock() - hit[0] < CACHE_SECONDS:
            out[name] = hit[1]
            continue
        try:
            value = fn(ctx, now)
        except Exception as err:  # one broken source must not blank the widget
            log.exception("collector %s", name)
            value = {"error": f"{type(err).__name__}: {err}"[:120]}
        _cache[name] = (clock(), value)
        out[name] = value
    out["level"] = level(out)
    return out


def make_ctx(env: dict) -> dict:
    home = Path.home()
    return {"env": env, "state_db": DB_PATH, "tasks_db": tasks.DB_PATH,
            "router_db": home / "9router" / "data" / "db" / "data.sqlite", "backups": home / "backups"}

<title>Hub</title><style>body{{font:15px system-ui;margin:16px;max-width:720px}}
td{{padding:2px 8px;vertical-align:top}} pre{{white-space:pre-wrap}}</style>{body}"""


def dashboard(s: dict, recent: list, token: str) -> str:
    rows = "".join(

        f"<td>{escape(t['title'])}</td><td>{t['status']}</td><td>{escape(t['model'] or '')}</td></tr>"
        for t in recent)
    sections = "".join(f"<h3>{name}</h3><pre>{escape(json.dumps(s[name], ensure_ascii=False, indent=1))}</pre>"
                       for name in COLLECTORS if name in s)
    return PAGE.format(body=f"<h2>{LEVEL_ICON[s['level']]} Hub · {s['updated']}</h2>"

def task_page(t: dict) -> str:
    return PAGE.format(body=f"<h2>#{t['id']} {escape(t['payload'].get('title') or t['kind'])}</h2>"
                            f"<p>{t['status']} · {escape(t['model'] or '-')} · {t['finished_at'] or ''}</p>"
                            f"<pre>{escape(t['result'] or t['error'] or '')}</pre>")


class Handler(BaseHTTPRequestHandler):
    ctx: dict = {}

    def _send(self, code: int, kind: str, body: str) -> None:
        data = body.encode()
        self.send_response(code)
        self.send_header("Content-Type", kind)
        self.send_header("Content-Length", str(len(data)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        url = urllib.parse.urlparse(self.path)
        given = urllib.parse.parse_qs(url.query).get("token", [""])[0]
        expected = self.ctx["env"].get("STATUS_TOKEN", "")
        if not expected or not hmac.compare_digest(given.encode(), expected.encode()):
            return self._send(403, "text/plain", "forbidden")
        now = datetime.now(timezone.utc)
        if url.path == "/status":
            return self._send(200, "application/json; charset=utf-8",
                              json.dumps(gather(self.ctx, now), ensure_ascii=False))
        with closing(tasks.connect(self.ctx["tasks_db"])) as db:
            if url.path == "/":
                return self._send(200, "text/html; charset=utf-8",
                                  dashboard(gather(self.ctx, now), tasks.recent(db), given))
            match = re.fullmatch(r"/task/(\d+)", url.path)
            task = tasks.get(db, int(match.group(1))) if match else None
        if task:
            return self._send(200, "text/html; charset=utf-8", task_page(task))
        return self._send(404, "text/plain", "not found")

    def log_message(self, fmt, *args):
        log.info("%s %s", self.address_string(), fmt % args)


def make_server(ctx: dict, port: int = PORT) -> ThreadingHTTPServer:
    handler = type("BoundHandler", (Handler,), {"ctx": ctx})
    return ThreadingHTTPServer(("127.0.0.1", port), handler)
```

```python
# hub/morning.py
"""Morning push text and the 'interview in 1 hour' check, used by infobot's scheduler."""
from __future__ import annotations

import urllib.error
from datetime import datetime, timedelta

from connectors import gcal
from connectors.google_auth import AuthError, TokenMissing
from hub.collectors import jobs, today
from hub.collectors.common import WIB


def morning_text(ctx: dict, now: datetime) -> str:
    agenda = today.collect(ctx, now)["items"]
    counts = jobs.collect(ctx, now)
    lines = [f"{item['time']} {item['title']}" if item["time"] else item["title"] for item in agenda]
    lines = lines or ["Tidak ada agenda hari ini."]
    lines.append(f"{counts['new_matches']} lowongan baru · {counts['active']} aktif · "
                 f"{counts['followup_due']} follow-up jatuh tempo")
    return "\n".join(lines)


def soon_events(ctx: dict, now: datetime, within: timedelta = timedelta(hours=1)) -> tuple:
    try:
        kwargs = {"token_fn": ctx["token_fn"]} if ctx.get("token_fn") else {}
        events = gcal.list_events(now.astimezone(WIB).date(), **kwargs)
    except TokenMissing:
        return [], None
    except (AuthError, urllib.error.URLError, OSError, ValueError) as err:
        return [], str(err)
    soon = [ev for ev in events
            if ev["start"] and now < datetime.fromisoformat(ev["start"]) <= now + within]
    return soon, None
```
